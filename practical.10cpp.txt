#include <iostream>
using namespace std;

int main() {
    int edge[6][3] = {
        {0, 1, 2},
        {0, 3, 6},
        {1, 2, 3},
        {1, 3, 8},
        {1, 4, 5},
        {2, 4, 7}
    };

    // Sort edges by weight
    for (int i = 0; i < 6; i++)
        for (int j = i + 1; j < 6; j++)
            if (edge[i][2] > edge[j][2])
                for (int k = 0; k < 3; k++)
                    swap(edge[i][k], edge[j][k]);

    int parent[5] = {0, 1, 2, 3, 4};
    int cost = 0, count = 0;

    for (int i = 0; i < 6 && count < 4; i++) {
        int u = edge[i][0];
        int v = edge[i][1];

        if (parent[u] != parent[v]) {
            cout << u << " - " << v
                 << " : " << edge[i][2] << endl;

            cost += edge[i][2];

            int old = parent[v];
            for (int j = 0; j < 5; j++)
                if (parent[j] == old)
                    parent[j] = parent[u];

            count++;
        }
    }

    cout << "Minimum Cost = " << cost;

    return 0;
}
